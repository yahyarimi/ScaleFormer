class MACCounter:

    def __init__(self):

        self.total_macs = 0

        self.breakdown = defaultdict(int)

        self.handles = []

        self.attention_calls = 0

        # Standard MultiheadAttention uses an out_proj
        # Linear object, but normally calls it through
        # torch.nn.functional rather than its module
        # forward. We count it explicitly in the MHA hook.
        self.mha_out_proj_ids = set()

    def add(
        self,
        category: str,
        macs: int,
    ):

        macs = int(macs)

        self.breakdown[
            category
        ] += macs

        self.total_macs += macs

    def linear_hook(
        self,
        module,
        args,
        kwargs,
        output,
    ):

        # Do not double-count an MHA output projection.
        if id(module) in self.mha_out_proj_ids:
            return

        if not isinstance(
            output,
            torch.Tensor,
        ):
            return

        # One MAC for each input feature contributing
        # to each output element.
        macs = (
            output.numel()
            * module.in_features
        )

        self.add(
            "linear",
            macs,
        )

    def conv1d_hook(
        self,
        module,
        args,
        kwargs,
        output,
    ):

        if not isinstance(
            output,
            torch.Tensor,
        ):
            return

        kernel = module.kernel_size[0]

        input_channels_per_group = (
            module.in_channels
            // module.groups
        )

        macs = (
            output.numel()
            * input_channels_per_group
            * kernel
        )

        self.add(
            "conv1d",
            macs,
        )

    def conv2d_hook(
        self,
        module,
        args,
        kwargs,
        output,
    ):

        if not isinstance(
            output,
            torch.Tensor,
        ):
            return

        kernel_h = module.kernel_size[0]
        kernel_w = module.kernel_size[1]

        input_channels_per_group = (
            module.in_channels
            // module.groups
        )

        macs = (
            output.numel()
            * input_channels_per_group
            * kernel_h
            * kernel_w
        )

        self.add(
            "conv2d",
            macs,
        )

    @staticmethod
    def sequence_information(
        tensor: torch.Tensor,
        batch_first: bool,
    ):

        if tensor.ndim != 3:
            raise RuntimeError(
                "Expected a 3D attention tensor, "
                f"got shape {tuple(tensor.shape)}."
            )

        if batch_first:

            batch = tensor.shape[0]
            length = tensor.shape[1]
            features = tensor.shape[2]

        else:

            length = tensor.shape[0]
            batch = tensor.shape[1]
            features = tensor.shape[2]

        return (
            int(batch),
            int(length),
            int(features),
        )

    def mha_hook(
        self,
        module,
        args,
        kwargs,
        output,
    ):

        self.attention_calls += 1

        if len(args) >= 3:

            query = args[0]
            key = args[1]
            value = args[2]

        else:

            query = kwargs.get(
                "query"
            )

            key = kwargs.get(
                "key"
            )

            value = kwargs.get(
                "value"
            )

        if (
            query is None
            or key is None
            or value is None
        ):
            raise RuntimeError(
                "Could not recover Q/K/V tensors "
                "from MultiheadAttention."
            )

        (
            batch_q,
            length_q,
            q_features,
        ) = self.sequence_information(
            query,
            module.batch_first,
        )

        (
            batch_k,
            length_k,
            k_features,
        ) = self.sequence_information(
            key,
            module.batch_first,
        )

        (
            batch_v,
            length_v,
            v_features,
        ) = self.sequence_information(
            value,
            module.batch_first,
        )

        if not (
            batch_q
            == batch_k
            == batch_v
        ):
            raise RuntimeError(
                "Attention batch sizes differ."
            )

        if length_k != length_v:
            raise RuntimeError(
                "Key/value sequence lengths differ."
            )

        batch = batch_q
        embed_dim = int(
            module.embed_dim
        )

        # ----------------------------------------------------
        # Q, K, V projections
        # ----------------------------------------------------

        q_projection = (
            batch
            * length_q
            * q_features
            * embed_dim
        )

        k_projection = (
            batch
            * length_k
            * k_features
            * embed_dim
        )

        v_projection = (
            batch
            * length_v
            * v_features
            * embed_dim
        )

        # ----------------------------------------------------
        # Q K^T
        #
        # Across all heads:
        # H * Lq * Lk * Dh
        # =
        # Lq * Lk * embed_dim
        # ----------------------------------------------------

        qk_macs = (
            batch
            * length_q
            * length_k
            * embed_dim
        )

        # ----------------------------------------------------
        # Attention probabilities times V
        # ----------------------------------------------------

        av_macs = (
            batch
            * length_q
            * length_k
            * embed_dim
        )

        # ----------------------------------------------------
        # Output projection
        # ----------------------------------------------------

        output_projection = (
            batch
            * length_q
            * embed_dim
            * embed_dim
        )

        self.add(
            "attention_q_projection",
            q_projection,
        )

        self.add(
            "attention_k_projection",
            k_projection,
        )

        self.add(
            "attention_v_projection",
            v_projection,
        )

        self.add(
            "attention_qk",
            qk_macs,
        )

        self.add(
            "attention_av",
            av_macs,
        )

        self.add(
            "attention_output_projection",
            output_projection,
        )

    def register(
        self,
        model: nn.Module,
    ):

        for module in model.modules():

            if isinstance(
                module,
                nn.MultiheadAttention,
            ):

                self.mha_out_proj_ids.add(
                    id(
                        module.out_proj
                    )
                )

        for module in model.modules():

            if isinstance(
                module,
                nn.MultiheadAttention,
            ):

                handle = (
                    module.register_forward_hook(
                        self.mha_hook,
                        with_kwargs=True,
                    )
                )

                self.handles.append(
                    handle
                )

            elif isinstance(
                module,
                nn.Linear,
            ):

                handle = (
                    module.register_forward_hook(
                        self.linear_hook,
                        with_kwargs=True,
                    )
                )

                self.handles.append(
                    handle
                )

            elif isinstance(
                module,
                nn.Conv1d,
            ):

                handle = (
                    module.register_forward_hook(
                        self.conv1d_hook,
                        with_kwargs=True,
                    )
                )

                self.handles.append(
                    handle
                )

            elif isinstance(
                module,
                nn.Conv2d,
            ):

                handle = (
                    module.register_forward_hook(
                        self.conv2d_hook,
                        with_kwargs=True,
                    )
                )

                self.handles.append(
                    handle
                )

    def remove(
        self,
    ):

        for handle in self.handles:

            handle.remove()

        self.handles.clear()


def count_model(
    name: str,
    model: nn.Module,
    parameters: int,
    interpolated: torch.Tensor,
    mask: torch.Tensor,
):

    print(
        "\n"
        + "=" * 80
    )

    print(
        f"COUNTING: {name.upper()}"
    )

    print(
        "=" * 80
    )

    counter = MACCounter()

    counter.register(
        model
    )

    try:

        with torch.inference_mode():

            # MACs are shape-dependent, not precision-dependent.
            # Use the same AMP path as inference.
            with torch.amp.autocast(
                device_type="cuda",
                dtype=torch.float16,
                enabled=True,
            ):

                output = model(
                    interpolated,
                    mask,
                )

        torch.cuda.synchronize()

    finally:

        counter.remove()

    if (
        output.shape
        != interpolated.shape
    ):

        raise RuntimeError(
            f"{name}: unexpected output shape."
        )

    total_macs = (
        counter.total_macs
    )

    # Conventional approximation:
    # one multiply-accumulate = two floating-point
    # operations (one multiplication + one addition).
    approximate_flops = (
        2
        * total_macs
    )

    print(
        f"Parameters          : "
        f"{parameters:,}"
    )

    print(
        f"Attention calls     : "
        f"{counter.attention_calls}"
    )

    for (
        category,
        value,
    ) in sorted(
        counter.breakdown.items()
    ):

        print(
            f"{category:<30}: "
            f"{value / 1e6:>12.4f} MMAC"
        )

    print(
        "-" * 80
    )

    print(
        f"TOTAL               : "
        f"{total_macs / 1e6:.4f} MMAC"
    )

    print(
        f"Approx. FLOPs       : "
        f"{approximate_flops / 1e9:.4f} GFLOPs"
    )

    return {
        "method": name,
        "parameters": parameters,
        "total_macs": (
            total_macs
        ),
        "mmacs": (
            total_macs
            / 1e6
        ),
        "approximate_flops": (
            approximate_flops
        ),
        "approximate_gflops": (
            approximate_flops
            / 1e9
        ),
        "attention_calls": (
            counter.attention_calls
        ),
        "breakdown": dict(
            counter.breakdown
        ),
    }

REPETITIONS = 3


def measure_repeated(
    function,
):

    peaks = []
    incremental_peaks = []

    for repetition in range(
        1,
        REPETITIONS + 1,
    ):

        result = measure_peak_memory(
            function
        )

        peaks.append(
            result[
                "peak_allocated_mib"
            ]
        )

        incremental_peaks.append(
            result[
                "incremental_peak_mib"
            ]
        )

        print(
            f"  repetition {repetition}: "
            f"peak={peaks[-1]:.4f} MiB | "
            f"incremental="
            f"{incremental_peaks[-1]:.4f} MiB"
        )

    return {
        "peak_mean_mib": (
            statistics.mean(
                peaks
            )
        ),
        "peak_std_mib": (
            statistics.stdev(
                peaks
            )
            if len(peaks) > 1
            else 0.0
        ),
        "peak_max_mib": max(
            peaks
        ),
        "incremental_peak_mean_mib": (
            statistics.mean(
                incremental_peaks
            )
        ),
    }

def percentile(
    values: list[float],
    fraction: float,
) -> float:

    ordered = sorted(values)

    index = int(
        fraction
        * (
            len(ordered)
            - 1
        )
    )

    return ordered[index]


def summarize(
    values: list[float],
) -> dict:

    return {
        "mean_ms": statistics.mean(
            values
        ),
        "std_ms": statistics.stdev(
            values
        ),
        "median_ms": statistics.median(
            values
        ),
        "min_ms": min(
            values
        ),
        "max_ms": max(
            values
        ),
        "p95_ms": percentile(
            values,
            0.95,
        ),
    }


def time_single_call(
    function,
) -> float:
    """
    Measure one GPU operation using CUDA events.

    Synchronizing each measurement avoids a long
    asynchronous timing queue and makes the paired
    measurements easier to interpret on a Windows
    laptop GPU.
    """

    start = torch.cuda.Event(
        enable_timing=True
    )

    end = torch.cuda.Event(
        enable_timing=True
    )

    start.record()

    output = function()

    end.record()

    end.synchronize()

    elapsed_ms = start.elapsed_time(
        end
    )

    del output

    return elapsed_ms

def benchmark_learned_model(
    name: str,
    model: torch.nn.Module,
    parameters: int,
    observed: torch.Tensor,
    mask: torch.Tensor,
):

    print(
        "\n"
        + "=" * 80
    )

    print(
        f"{name.upper()} LATENCY"
    )

    print(
        "=" * 80
    )

    interpolated = (
        frequency_linear_interpolate(
            observed,
            mask,
        )
    )

    torch.cuda.synchronize()

    # Verify that recomputing interpolation gives
    # exactly the same tensor properties.
    probe = (
        frequency_linear_interpolate(
            observed,
            mask,
        )
    )

    max_difference = (
        probe
        .sub(
            interpolated
        )
        .abs()
        .max()
        .item()
    )

    print(
        f"Interpolation check : "
        f"max diff "
        f"{max_difference:.8e}"
    )

    print(
        f"Interpolation dtype : "
        f"{interpolated.dtype}"
    )

    print(
        f"Interpolation shape : "
        f"{tuple(interpolated.shape)}"
    )

    print(
        f"Contiguous          : "
        f"{interpolated.is_contiguous()}"
    )

    del probe

    def model_only():

        with torch.amp.autocast(
            device_type="cuda",
            dtype=torch.float16,
            enabled=True,
        ):

            return model(
                interpolated,
                mask,
            )

    def end_to_end():

        current_interpolated = (
            frequency_linear_interpolate(
                observed,
                mask,
            )
        )

        with torch.amp.autocast(
            device_type="cuda",
            dtype=torch.float16,
            enabled=True,
        ):

            return model(
                current_interpolated,
                mask,
            )

    # --------------------------------------------------------
    # Paired warm-up
    #
    # Alternate order so neither path systematically benefits
    # from being measured after the other.
    # --------------------------------------------------------

    with torch.inference_mode():

        for index in range(
            WARMUP_PAIRS
        ):

            if index % 2 == 0:

                _ = model_only()
                _ = end_to_end()

            else:

                _ = end_to_end()
                _ = model_only()

        torch.cuda.synchronize()

        model_times = []
        e2e_times = []

        # ----------------------------------------------------
        # Paired/interleaved measurements
        # ----------------------------------------------------

        for index in range(
            MEASURED_PAIRS
        ):

            if index % 2 == 0:

                model_ms = (
                    time_single_call(
                        model_only
                    )
                )

                e2e_ms = (
                    time_single_call(
                        end_to_end
                    )
                )

            else:

                e2e_ms = (
                    time_single_call(
                        end_to_end
                    )
                )

                model_ms = (
                    time_single_call(
                        model_only
                    )
                )

            model_times.append(
                model_ms
            )

            e2e_times.append(
                e2e_ms
            )

    model_stats = summarize(
        model_times
    )

    e2e_stats = summarize(
        e2e_times
    )

    print()

    print(
        f"Parameters          : "
        f"{parameters:,}"
    )

    print(
        f"Model-only mean     : "
        f"{model_stats['mean_ms']:.4f} "
        f"± "
        f"{model_stats['std_ms']:.4f} ms"
    )

    print(
        f"Model-only median   : "
        f"{model_stats['median_ms']:.4f} ms"
    )

    print(
        f"Model-only P95      : "
        f"{model_stats['p95_ms']:.4f} ms"
    )

    print(
        f"End-to-end mean     : "
        f"{e2e_stats['mean_ms']:.4f} "
        f"± "
        f"{e2e_stats['std_ms']:.4f} ms"
    )

    print(
        f"End-to-end median   : "
        f"{e2e_stats['median_ms']:.4f} ms"
    )

    print(
        f"End-to-end P95      : "
        f"{e2e_stats['p95_ms']:.4f} ms"
    )

    difference = (
        e2e_stats[
            "median_ms"
        ]
        - model_stats[
            "median_ms"
        ]
    )

    print(
        f"Median E2E overhead : "
        f"{difference:.4f} ms"
    )

    if (
        e2e_stats[
            "median_ms"
        ]
        <= model_stats[
            "median_ms"
        ]
    ):

        print(
            "WARNING: end-to-end median "
            "is not greater than "
            "model-only median."
        )

    return {
        "method": name,
        "parameters": (
            parameters
        ),
        "model_only_mean_ms": (
            model_stats[
                "mean_ms"
            ]
        ),
        "model_only_std_ms": (
            model_stats[
                "std_ms"
            ]
        ),
        "model_only_median_ms": (
            model_stats[
                "median_ms"
            ]
        ),
        "model_only_p95_ms": (
            model_stats[
                "p95_ms"
            ]
        ),
        "end_to_end_mean_ms": (
            e2e_stats[
                "mean_ms"
            ]
        ),
        "end_to_end_std_ms": (
            e2e_stats[
                "std_ms"
            ]
        ),
        "end_to_end_median_ms": (
            e2e_stats[
                "median_ms"
            ]
        ),
        "end_to_end_p95_ms": (
            e2e_stats[
                "p95_ms"
            ]
        ),
        "median_e2e_overhead_ms": (
            difference
        ),
    }

def aggregate(
    rows: list[dict],
):

    methods = sorted(
        {
            row["method"]
            for row in rows
        }
    )

    summary_rows = []

    for method in methods:

        method_rows = [
            row
            for row in rows
            if row["method"] == method
        ]

        e2e_medians = [
            row[
                "end_to_end_median_ms"
            ]
            for row in method_rows
        ]

        e2e_p95 = [
            row[
                "end_to_end_p95_ms"
            ]
            for row in method_rows
        ]

        model_values = [
            row[
                "model_only_median_ms"
            ]
            for row in method_rows
            if row[
                "model_only_median_ms"
            ] is not None
        ]

        row = {
            "method": method,
            "rounds": len(
                method_rows
            ),
            "parameters": (
                method_rows[
                    0
                ][
                    "parameters"
                ]
            ),
            "e2e_median_mean_ms": (
                statistics.mean(
                    e2e_medians
                )
            ),
            "e2e_median_std_ms": (
                statistics.stdev(
                    e2e_medians
                )
            ),
            "e2e_median_min_ms": (
                min(
                    e2e_medians
                )
            ),
            "e2e_median_max_ms": (
                max(
                    e2e_medians
                )
            ),
            "e2e_p95_mean_ms": (
                statistics.mean(
                    e2e_p95
                )
            ),
            "e2e_p95_std_ms": (
                statistics.stdev(
                    e2e_p95
                )
            ),
            "model_median_mean_ms": (
                statistics.mean(
                    model_values
                )
                if model_values
                else None
            ),
            "model_median_std_ms": (
                statistics.stdev(
                    model_values
                )
                if len(
                    model_values
                ) > 1
                else None
            ),
        }

        summary_rows.append(
            row
        )

    return summary_rows


def save_csv(
    rows: list[dict],
    path: Path,
):

    if not rows:
        return

    fieldnames = []

    for row in rows:
        for key in row:
            if key not in fieldnames:
                fieldnames.append(
                    key
                )

    with path.open(
        "w",
        newline="",
        encoding="utf-8",
    ) as file:

        writer = csv.DictWriter(
            file,
            fieldnames=fieldnames,
            extrasaction="ignore",
        )

        writer.writeheader()

        writer.writerows(
            rows
        )
